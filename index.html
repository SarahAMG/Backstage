import { serve } from "https://deno.land/std@0.168.0/http/server.ts"; import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const AMG_URL = "https://www.academymusicgroup.com/api/search/events"; const WANTED_VENUES = [ "O2 Forum Kentish Town", "O2 Academy Islington", "O2 Shepherd's Bush Empire", "O2 Academy Brixton", "O2 Academy Bristol", "O2 Guildhall Southampton", "O2 Academy Bournemouth", "O2 Academy Leeds", "O2 Academy Birmingham", "O2 Institute Birmingham", "O2 Academy Liverpool", "O2 Academy Oxford", "O2 Academy Glasgow", "O2 Academy Sheffield", "O2 Apollo Manchester", "O2 Victoria Warehouse Manchester", "O2 Ritz Manchester", "O2 Academy Leicester", "O2 City Hall Newcastle", ]; const VENUE_ALIASES: Record = { "o2 academy2 sheffield": "O2 Academy Sheffield", }; function normalise(value: string | null | undefined) { return (value || "").trim().toLowerCase(); } function firstString(...values: any[]) { for (const value of values) { if (typeof value === "string" && value.trim()) { return value.trim(); } } return null; } function firstDate(...values: any[]) { for (const value of values) { if (!value) continue; if (typeof value === "string" && value.trim()) { const parsed = new Date(value); if (!Number.isNaN(parsed.getTime())) { return parsed.toISOString(); } } } return null; } function findImage(event: any) { const possibleImages = [ event.image, event.imageUrl, event.imageURL, event.image_url, event.eventImage, event.eventImageUrl, event.eventImageURL, event.event_image, event.event_image_url, event.artwork, event.artworkUrl, event.artworkURL, event.artwork_url, event.poster, event.posterUrl, event.posterURL, event.poster_url, event.heroImage, event.heroImageUrl, event.hero_image, event.media?.image, event.media?.imageUrl, event.media?.artwork, ]; const direct = firstString(...possibleImages); if (direct) { return direct; } if (Array.isArray(event.images)) { const image = event.images.find((item: any) => { return ( typeof item === "string" || item?.url || item?.src || item?.imageUrl ); }); if (typeof image === "string") { return image; } if (image) { return firstString( image.url, image.src, image.imageUrl, image.imageURL, ); } } return null; } function findTicketStatus(event: any) { const possibleStatuses = [ event.ticketStatus, event.ticket_status, event.status, event.eventStatus, event.event_status, event.ticketAvailability, event.ticketAvailabilityStatus, event.availability, ]; const status = firstString(...possibleStatuses); if (!status) { return null; } const lower = status.toLowerCase(); if (lower.includes("cancel")) { return "Cancelled"; } if (lower.includes("sold out") || lower.includes("soldout")) { return "Sold Out"; } if (lower.includes("low")) { return "Low Tickets"; } if ( lower.includes("on sale") || lower.includes("onsale") || lower.includes("available") ) { return "On Sale"; } if ( lower.includes("coming") || lower.includes("future") || lower.includes("not on sale") ) { return "Coming Soon"; } return status; } function findSaleAccess(event: any) { const possibleValues = [ event.saleAccess, event.sale_access, event.ticketAccess, event.ticket_access, event.saleType, event.sale_type, event.ticketType, event.ticket_type, ]; const value = firstString(...possibleValues); if (value) { const lower = value.toLowerCase(); if (lower.includes("priority")) { return "O2 Priority"; } if (lower.includes("general")) { return "General Sale"; } } // Look through ticket records as well if (Array.isArray(event.tickets)) { const priorityTicket = event.tickets.find((ticket: any) => { const text = JSON.stringify(ticket).toLowerCase(); return text.includes("priority"); }); if (priorityTicket) { return "O2 Priority"; } const generalTicket = event.tickets.find((ticket: any) => { const text = JSON.stringify(ticket).toLowerCase(); return text.includes("general sale"); }); if (generalTicket) { return "General Sale"; } } return null; } function findPrioritySaleDate(event: any) { return firstDate( event.prioritySaleDate, event.prioritySaleDateUtc, event.priority_sale_date, event.priority_sale_date_utc, event.o2PriorityDate, event.o2PriorityDateUtc, event.o2_priority_date, event.o2_priority_date_utc, event.presaleDate, event.presaleDateUtc, event.preSaleDate, event.preSaleDateUtc, ); } function findGeneralSaleDate(event: any) { return firstDate( event.generalSaleDate, event.generalSaleDateUtc, event.general_sale_date, event.general_sale_date_utc, event.publicSaleDate, event.publicSaleDateUtc, event.public_sale_date, event.public_sale_date_utc, event.onSaleDate, event.onSaleDateUtc, event.onsaleDate, event.onsaleDateUtc, ); } serve(async () => { try { const supabase = createClient( Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!, ); // Load the 19 venues const { data: dbVenues, error: venueError } = await supabase .from("venues") .select("id, name"); if (venueError) { throw new Error(`Could not load venues: ${venueError.message}`); } const wantedMap = new Map(); for (const venue of dbVenues || []) { const name = normalise(venue.name); if (WANTED_VENUES.some((v) => normalise(v) === name)) { wantedMap.set(name, venue); } } // Sheffield alias const sheffieldVenue = (dbVenues || []).find( (v: any) => normalise(v.name) === "o2 academy sheffield", ); if (sheffieldVenue) { wantedMap.set("o2 academy2 sheffield", sheffieldVenue); } // Fetch AMG events const pages = Array.from({ length: 20 }, (_, i) => i + 1); const responses = await Promise.all( pages.map(async (page) => { const response = await fetch( `${AMG_URL}?Page=${page}&PageSize=500`, { headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/131 Safari/537.36", "Accept": "application/json", }, }, ); if (!response.ok) { throw new Error( `AMG page ${page} returned HTTP ${response.status}`, ); } return response.json(); }), ); const allEvents = responses.flatMap( (data: any) => data.documents || [], ); // Only keep our 19 venues const importedEvents = allEvents.filter((event: any) => { const venueName = normalise(event.venue?.name); return ( wantedMap.has(venueName) || WANTED_VENUES.some((v) => normalise(v) === venueName) ); }); // Build shows const shows = importedEvents .map((event: any) => { const amgVenueName = event.venue?.name || ""; const normalisedAmgName = normalise(amgVenueName); const canonicalName = VENUE_ALIASES[normalisedAmgName] || amgVenueName; const dbVenue = wantedMap.get(normalise(canonicalName)) || wantedMap.get(normalisedAmgName); if (!dbVenue) return null; // Headline artist const headline = event.lineup?.find( (artist: any) => artist.type === "headline", ) || event.lineup?.find( (artist: any) => artist.isPrimary === true, ); const artist = headline?.name || event.lineup?.[0]?.name || event.name || "Unknown Artist"; // Ticket URL const ticketUrl = event.tickets?.find( (ticket: any) => ticket.ticketUrl, )?.ticketUrl || null; // Event URL const eventUrl = event.localizations?.find( (localisation: any) => localisation.url, )?.url || null; // Local event date const eventDate = event.eventDate?.split("T")[0] || event.eventDateUtc?.split("T")[0] || null; // New information const imageUrl = findImage(event); const ticketStatus = findTicketStatus(event); const saleAccess = findSaleAccess(event); const prioritySaleDate = findPrioritySaleDate(event); const generalSaleDate = findGeneralSaleDate(event); return { id: Number(event.id), ticketmaster_discovery_id: event.ticketmasterDiscoveryId || null, venue_id: dbVenue.id, artist, show_title: event.name || artist, city: event.venue?.city || null, event_date: eventDate, event_time: event.showTime || event.doorTime || null, genre: event.genres ?.map((genre: any) => genre.name) .filter(Boolean) .join(", ") || null, ticket_url: ticketUrl, event_url: eventUrl, image_url: imageUrl, ticket_status: ticketStatus, sale_access: saleAccess, priority_sale_date: prioritySaleDate, general_sale_date: generalSaleDate, last_seen: new Date().toISOString(), updated_at: new Date().toISOString(), }; }) .filter(Boolean); // Remove duplicate event IDs const uniqueShows = Array.from( new Map( shows.map((show: any) => [show.id, show]), ).values(), ); const ids = uniqueShows.map((show: any) => show.id); // Check existing shows in batches const existingShows: any[] = []; for (let i = 0; i < ids.length; i += 500) { const idBatch = ids.slice(i, i + 500); const { data, error } = await supabase .from("shows") .select("id") .in("id", idBatch); if (error) { throw new Error( `Could not check existing shows: ${error.message}`, ); } if (data) { existingShows.push(...data); } } const existingIds = new Set( existingShows.map((show: any) => show.id), ); // New shows const newShows = uniqueShows .filter((show: any) => !existingIds.has(show.id)) .map((show: any) => ({ ...show, first_seen: new Date().toISOString(), })); // Existing shows const existingUpdates = uniqueShows.filter( (show: any) => existingIds.has(show.id), ); // Insert new shows if (newShows.length > 0) { const { error } = await supabase .from("shows") .insert(newShows); if (error) { throw new Error( `Could not insert shows: ${error.message}`, ); } } // Update existing shows // Does NOT touch show_marketing. for (const show of existingUpdates) { const { id, ...updateData } = show; const { error } = await supabase .from("shows") .update(updateData) .eq("id", id); if (error) { throw new Error( `Could not update show ${id}: ${error.message}`, ); } } return new Response( JSON.stringify( { success: true, venues: WANTED_VENUES.length, amgEventsScanned: allEvents.length, matchingEvents: importedEvents.length, uniqueShows: uniqueShows.length, newShows: newShows.length, updatedShows: existingUpdates.length, }, null, 2, ), { headers: { "Content-Type": "application/json", }, }, ); } catch (error) { return new Response( JSON.stringify( { success: false, error: error instanceof Error ? error.message : String(error), }, null, 2, ), { status: 500, headers: { "Content-Type": "application/json", }, }, ); } });
